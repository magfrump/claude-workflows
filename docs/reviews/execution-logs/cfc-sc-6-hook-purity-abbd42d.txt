--- [G] APPENDIX: the hook-purity harness, verbatim ---
set -u
R=$(mktemp -d); export CMD_LOG="$R/cmds.log"; : > "$CMD_LOG"
export CC_EGRESS_DIR=/workspace/devcontainer-config/egress
export CC_EGRESS_PROFILE_FILE="$R/profile"; : > "$CC_EGRESS_PROFILE_FILE"
export CC_DNSMASQ_CONF="$R/dnsmasq.d/cc-allowlist.conf"
export CC_DNSMASQ_PIDFILE="$R/cc-dnsmasq.pid"
export CC_SNI_RUN_DIR="$R/cc-sni-proxy"
S="$R/bin"; mkdir -p "$S"
export CC_SNI_PROXY_BIN="$S/cc-sni-proxy"
for b in iptables iptables-save ipset ip dig curl dnsmasq pkill runuser cc-sni-proxy aggregate; do
  printf '#!/usr/bin/env bash\necho "%s $*" >> "$CMD_LOG"\nexit 0\n' "$b" > "$S/$b"
done
chmod +x "$S"/*
export PATH="$S:$PATH"
printf 'nameserver 10.9.8.7\n' > "$R/resolv.conf"
for h in --print-domains --print-entries; do
  echo "### $h"; bash /workspace/devcontainer-config/init-firewall.sh "$h"; echo "exit=$?"
done
for h in --print-resolvers --print-dnsmasq-conf; do
  echo "### $h $R/resolv.conf"; bash /workspace/devcontainer-config/init-firewall.sh "$h" "$R/resolv.conf"; echo "exit=$?"
  echo "### $h (no 2nd arg -> defaults to /etc/resolv.conf)"; bash /workspace/devcontainer-config/init-firewall.sh "$h" >/dev/null; echo "exit=$?"
done
echo "### side effects"
echo "CMD_LOG bytes: $(wc -c < "$CMD_LOG")  (0 == no stubbed binary was invoked)"
echo "dnsmasq conf exists: $([ -e "$CC_DNSMASQ_CONF" ] && echo yes || echo no)"
echo "dnsmasq pidfile exists: $([ -e "$CC_DNSMASQ_PIDFILE" ] && echo yes || echo no)"
echo "SNI run dir exists: $([ -e "$CC_SNI_RUN_DIR" ] && echo yes || echo no)"
echo "files created under \$R besides the inputs:"; ls -A "$R"
rm -rf "$R"
