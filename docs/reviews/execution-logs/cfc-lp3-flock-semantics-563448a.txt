# Q: does bash 'flock -u N' release the advisory lock while keeping fd N open?
# Q: is a per-command '9>&-' redirection scoped to that command only?
$ bash <<'SCRIPT'
set -uo pipefail
flock --version; L=$(mktemp); exec 9>"$L"
flock -w 5 9                       # this shell takes the lock
flock -n "$L" true                 # outsider, expect rc=1
flock -u 9                         # release
flock -n "$L" true                 # outsider, expect rc=0
[ -e /proc/$$/fd/9 ]; echo hi >&9  # fd still open+writable?
exec 9>"$L"; bash -c '[ -e /proc/self/fd/9 ]' 9>&-; [ -e /proc/$$/fd/9 ]
SCRIPT
