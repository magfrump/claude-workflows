#!/usr/bin/env bash
# code-fact-check probe: old vs new hook, tainted session, file tools
set -u
S=/tmp/claude-1000/-workspace/104b63ce-e414-465c-a24f-dda1e4116218/scratchpad
export CC_WEB_TAINT_DIR="$S/cfc_taint"
mkdir -p "$CC_WEB_TAINT_DIR"; : > "$CC_WEB_TAINT_DIR/cfcsid"
NEW=/workspace/hooks/guard-trusted-writes.py
OLD="$S/cfc_old_guard.py"
git -C /workspace show e8d5fa1:hooks/guard-trusted-writes.py > "$OLD"
C="CLAUDE"".md"
run() {  # tool, path
  local n o
  n=$(printf '{"tool_name":"%s","session_id":"cfcsid","tool_input":{"file_path":"%s"}}' "$1" "$2" | python3 "$NEW")
  o=$(printf '{"tool_name":"%s","session_id":"cfcsid","tool_input":{"file_path":"%s"}}' "$1" "$2" | python3 "$OLD")
  n=$(echo "${n:-defer}" | grep -o '"permissionDecision": "[a-z]*"' || echo defer)
  o=$(echo "${o:-defer}" | grep -o '"permissionDecision": "[a-z]*"' || echo defer)
  printf '%-6s %-70s NEW=%s OLD=%s\n' "$1" "$2" "$n" "$o"
}
echo "HOME=$HOME"; ls -la "$HOME/.claude" | grep -E 'hooks|CLAUDE|settings'
readlink -f "$HOME/.claude/hooks"
run Edit "$HOME/.claude/hooks/guard-trusted-writes.py"
run Edit "$HOME/.claude/x/../hooks/guard-trusted-writes.py"
run Edit "$HOME/.claude/./hooks/guard-trusted-writes.py"
run Edit "$HOME/.claude//hooks/guard-trusted-writes.py"
run Edit "$HOME/.claude/$C"
run Edit "$HOME/.claude/x/../$C"
run Edit "$HOME/.claude/settings.json"
run Edit "$HOME/.claude/x/../settings.json"
run Edit "/opt/claude-workflows/hooks/guard-trusted-writes.py"
P="$S/cfc_proj"; rm -rf "$P"; mkdir -p "$P"; ln -s "$HOME/.claude" "$P/.claude"
run Edit "$P/.claude/hooks/guard-trusted-writes.py"
run Edit "$P/.claude/settings.json"
run Edit "$P/.claude/$C"
run Edit "/workspace/.claude/settings.json"
run Edit "/workspace/.claude/hooks/x.py"
run Edit "/workspace/$C"
echo "--- untainted"
n=$(printf '{"tool_name":"Edit","session_id":"nosid","tool_input":{"file_path":"%s"}}' "/workspace/.claude/settings.json" | python3 "$NEW"); echo "untainted project settings -> ${n:-defer}"
echo "--- relative CLAUDE_CONFIG_DIR"
mkdir -p "$S/cfc_cfg"
(cd "$S" && CLAUDE_CONFIG_DIR=cfc_cfg python3 -c "
import importlib.util;spec=importlib.util.spec_from_file_location('g','$NEW');g=importlib.util.module_from_spec(spec);spec.loader.exec_module(g)
print('cwd=scratch', g.GLOBAL_DIRS, g.classify_path('$S/cfc_cfg/settings.json'))")
(cd /tmp && CLAUDE_CONFIG_DIR=cfc_cfg python3 -c "
import importlib.util;spec=importlib.util.spec_from_file_location('g','$NEW');g=importlib.util.module_from_spec(spec);spec.loader.exec_module(g)
print('cwd=/tmp', g.GLOBAL_DIRS, g.classify_path('$S/cfc_cfg/settings.json'))")
(CLAUDE_CONFIG_DIR="$S/cfc_cfg" python3 -c "
import importlib.util;spec=importlib.util.spec_from_file_location('g','$NEW');g=importlib.util.module_from_spec(spec);spec.loader.exec_module(g)
print('abs cfg: home settings', g.classify_path('$HOME/.claude/settings.json'), 'cfg settings', g.classify_path('$S/cfc_cfg/settings.json'), 'cfg hooks', g.classify_path('$S/cfc_cfg/hooks/a'))")
