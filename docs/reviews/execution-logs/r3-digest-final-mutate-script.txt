#!/usr/bin/env bash
# Mutation sweep: each mutant breaks the behaviour one test names; run that test only.
set -u
W=/workspace/.claude/wt-digest
M=/tmp/claude-1000/-workspace/fba1bfc0-301f-41fe-8500-89690af4d400/scratchpad/r3df/mut
export LC_ALL=C
mut() { # $1 test-number  $2 bats -f filter  $3 description  $4 perl expr
  rm -rf "$M"; mkdir -p "$M/scripts" "$M/test/scripts"
  cp "$W/scripts/dev-cycle.sh" "$W/scripts/questions.sh" "$M/scripts/"
  cp "$W/test/scripts/dev-cycle.bats" "$M/test/scripts/"
  perl -0pi -e "$4" "$M/scripts/dev-cycle.sh"
  if cmp -s "$W/scripts/dev-cycle.sh" "$M/scripts/dev-cycle.sh"; then echo "T$1 MUTATION DID NOT APPLY: $3"; return; fi
  out=$(timeout 120 bats -f "$2" "$M/test/scripts/dev-cycle.bats" 2>&1 | grep -v setlocale)
  if grep -q '^not ok' <<< "$out"; then r=KILLED; elif grep -q '^ok' <<< "$out"; then r=SURVIVED; else r="NO-RUN"; fi
  echo "T$1 $r — $3"
}
