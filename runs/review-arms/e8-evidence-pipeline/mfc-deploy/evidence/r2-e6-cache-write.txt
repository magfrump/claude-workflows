// Temporary code-fact-check probe (replicate r2) — verifies the LLM cache
// writes to the local filesystem at data/cache/<hash>.json under process.cwd().
// This file is deleted after the run.
import { describe, it, expect } from "vitest";
import { existsSync, readFileSync } from "fs";
import { join } from "path";
import { computeHash, setCachedResult, getCachedResult, removeCachedResult } from "./cache";

describe("cfc-r2 cache filesystem write probe", () => {
  it("setCachedResult writes data/cache/<hash>.json to the local filesystem", async () => {
    const model = "cfc-r2-probe-model";
    const sys = "probe-system";
    const user = "probe-user";
    const maxTokens = 7;
    const hash = computeHash(model, sys, user, maxTokens);
    const filePath = join(process.cwd(), "data", "cache", `${hash}.json`);

    await setCachedResult(hash, {
      text: "probe-text",
      usage: {
        provider: "mock", model, inputTokens: 1, outputTokens: 1,
        costUsd: 0, latencyMs: 0,
      },
      cacheKey: { model, systemPrompt: sys, userContent: user, maxTokens },
    } as never);

    expect(existsSync(filePath)).toBe(true);
    const onDisk = JSON.parse(readFileSync(filePath, "utf-8"));
    expect(onDisk.text).toBe("probe-text");

    const roundTrip = await getCachedResult(model, sys, user, maxTokens);
    expect(roundTrip?.text).toBe("probe-text");

    await removeCachedResult(hash); // clean up the cache file
    expect(existsSync(filePath)).toBe(false);
  });
});
