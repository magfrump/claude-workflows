390	# Checked HERE, before the flush, so a broken image (no binary, no service user)
391	# aborts with the live ruleset intact instead of failing closed mid-rebuild.
392	# The two paths are env-overridable for the unit tests only: this script runs via

396	DNSMASQ_PIDFILE="${CC_DNSMASQ_PIDFILE:-/run/cc-dnsmasq.pid}"

401	# Numeric, so the iptables owner rules never depend on a name lookup at rule time,
402	# and never 0: the whole design rests on dnsmasq NOT being root.
403	DNSMASQ_UID="$(id -u dnsmasq 2>/dev/null || true)"

431	# Same stance as dnsmasq: every "is the machinery there" check runs in phase A so a
