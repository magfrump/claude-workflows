#!/usr/bin/env bash
set -u
DC=/workspace/.claude/wt-digest/scripts/dev-cycle.sh
S=/tmp/claude-1000/-workspace/fba1bfc0-301f-41fe-8500-89690af4d400/scratchpad/r3df/q
rm -rf "$S"; mkdir -p "$S"
export GIT_CONFIG_GLOBAL=/dev/null GIT_CONFIG_NOSYSTEM=1 LC_ALL=C
export GIT_AUTHOR_NAME=t GIT_AUTHOR_EMAIL=t@x.invalid GIT_COMMITTER_NAME=t GIT_COMMITTER_EMAIL=t@x.invalid
export HOME="$S/home"; mkdir -p "$HOME"
mk() { mkdir -p "$1"; ( cd "$1"; git init -q -b "$2"; git commit -q --allow-empty -m initial
  for n in $(seq 1 $3); do git checkout -q -b f$n; git commit -q --allow-empty -m "feature $n"; git checkout -q "$2"; git merge -q --no-ff f$n -m "merge $n"; done ); }

echo "=== Q1 record added on a branch before the cycle-record date, merged after it"
mk "$S/q1" main 1; cd "$S/q1"
mkdir -p docs/working/cycles
git checkout -q -b late
mkdir -p docs/decisions; printf '# 003\n\n## Revisit triggers\nif branch-authored-early.\n' > docs/decisions/003-late.md
git add docs/decisions; GIT_COMMITTER_DATE=2026-03-05T12:00:00 git commit -q --date=2026-03-05T12:00:00 -m "add 003 on branch"
git checkout -q main; git merge -q --no-ff late -m "merge late (now)"
touch docs/working/cycles/cycle-2026-03-10.md
timeout 30 bash "$DC" | sed -n '/## 1/,/## 3/p'; echo "[exit ${PIPESTATUS[0]}]"

echo "=== Q2 sample: same day same, different days (3 merges, sample 2) over 30 days"
mk "$S/q2" main 3; cd "$S/q2"
a=$(DEV_CYCLE_TODAY=2026-05-01 timeout 30 bash "$DC" --since=2000-01-01 | sed -n '/## 4/,/## 5/p')
b=$(DEV_CYCLE_TODAY=2026-05-01 timeout 30 bash "$DC" --since=2000-01-01 | sed -n '/## 4/,/## 5/p')
[ "$a" = "$b" ] && echo "same day: identical" || echo "same day: DIFFER"
for i in $(seq 0 29); do d=$(date -d "2026-05-01 +$i day" +%F); DEV_CYCLE_TODAY=$d timeout 30 bash "$DC" --since=2000-01-01 | sed -n '/## 4/,/## 5/p' | grep '^- ' | awk '{print $NF}' | sort | tr '\n' ,; echo; done | sort | uniq -c
echo "--- adjacent days identical count (3 merges, sample 2):"
prev=""; same=0; for i in $(seq 0 29); do d=$(date -d "2026-05-01 +$i day" +%F); cur=$(DEV_CYCLE_TODAY=$d timeout 30 bash "$DC" --since=2000-01-01 | sed -n '/## 4/,/## 5/p' | grep '^- ' | sort | tr '\n' ,); [ "$cur" = "$prev" ] && same=$((same+1)); prev=$cur; done; echo "$same of 29 adjacent pairs identical"
echo "--- raw-date seeding (old approach) across 4 dates, sample 3 of 20 merges"
mk "$S/q3" main 20; cd "$S/q3"
for d in 2026-01-01 2026-01-15 2026-02-01 2026-03-01; do git log --merges --format=%s | shuf -n 3 --random-source=<(yes "$d") | sort | tr '\n' ,; echo; done | sort | uniq -c
echo "--- raw-date seeding, dates in different decades"
for d in 1999-01-01 2026-01-01 3000-12-31; do git log --merges --format=%s | shuf -n 3 --random-source=<(yes "$d") | sort | tr '\n' ,; echo; done | sort | uniq -c

echo "=== Q4 questions.sh open output (read-only, the worktree's own questions doc)"
cd /workspace/.claude/wt-digest && timeout 30 bash scripts/questions.sh open | head -5 | cat -A | head -5
