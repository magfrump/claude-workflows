235	def daemonize(args):
236	    """Fork, detach, drop to --user, bind, then report readiness to the caller.
237	    Exit status is the contract init-firewall.sh relies on: 0 only once the
238	    child is LISTENING; anything else means "no proxy", and the firewall must
239	    then fail closed rather than install a REDIRECT to nothing."""
240	    if os.geteuid() == 0 and not args.user:
241	        sys.exit("cc-sni-proxy: refusing to run as root; pass --user")
242	    stop_prior(args.pidfile)
243	    log_fd = os.open(args.log, os.O_WRONLY | os.O_CREAT | os.O_TRUNC, 0o644)
244	    r, w = os.pipe()
245	    pid = os.fork()
246	    if pid:
247	        os.close(w)
248	        os.close(log_fd)
249	        msg = os.read(r, 4096)
250	        if msg == b"ready":
251	            with open(args.pidfile, "w") as f:
252	                f.write(f"{pid}\n")
253	            return 0
254	        os.waitpid(pid, 0)
255	        sys.stderr.write(f"cc-sni-proxy: failed to start: {msg.decode(errors='replace') or 'child exited'}\n")
256	        return 1
